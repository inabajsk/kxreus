# EusView デスクトップ版の物理と全身の QP: odesim + wbqp (iOS 版と同じ) + JNI (Android 版と同じ odesim_jni.cpp, wbqp_jni.cpp) → libeusviewode.{so,dylib}
#   ソースは eusview/android/app/src/main/cpp/ (odesim.{h,cpp} は ios/EusView/Physics/, wbqp.{h,cpp} は ios/EusView/QP/ のコピー)
#   ODE (0.16, 倍精度) の見つけ方:
#     -DODE_ROOT=<dir>   <dir>/include/ode/ode.h と <dir>/lib/libode.a (make ode で作る静的ライブラリ)
#     macOS で ODE_ROOT がなければ eusview/ios/third_party/ODE.xcframework/macos-x86_64 (Intel Mac だけ. Apple Silicon は make ode)
#     Linux で ODE_ROOT がなければ pkg-config ode (sudo apt install libode-dev)
cmake_minimum_required(VERSION 3.16)
project(eusviewode CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(EUSVIEW ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(SRC ${EUSVIEW}/android/app/src/main/cpp)
# JNI のヘッダ: $JAVA_HOME/include (headless の JDK でもよいように FindJNI は JAVA_HOME がないときだけ)
if(DEFINED ENV{JAVA_HOME} AND EXISTS "$ENV{JAVA_HOME}/include/jni.h")
  if(APPLE)
    set(JNI_INCLUDE_DIRS "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/darwin")
  else()
    set(JNI_INCLUDE_DIRS "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/linux")
  endif()
else()
  find_package(JNI REQUIRED)
endif()
message(STATUS "JNI: ${JNI_INCLUDE_DIRS}")

add_library(eusviewode SHARED ${SRC}/odesim.cpp ${SRC}/odesim_jni.cpp ${SRC}/wbqp.cpp ${SRC}/wbqp_jni.cpp)
target_include_directories(eusviewode PRIVATE ${SRC} ${JNI_INCLUDE_DIRS})
target_compile_options(eusviewode PRIVATE -O2)

set(ODE_ROOT "" CACHE PATH "ODE (include/ と lib/libode.a)")
if(NOT ODE_ROOT AND APPLE)
  if(CMAKE_HOST_SYSTEM_PROCESSOR STREQUAL "arm64")
    message(FATAL_ERROR "ios/third_party/ODE.xcframework の Mac 用は x86_64 だけです. make ode で ODE を作ってください")
  endif()
  set(ODE_ROOT ${EUSVIEW}/ios/third_party/ODE.xcframework/macos-x86_64)
  set(ODE_INC ${ODE_ROOT}/Headers)
  set(ODE_LIB ${ODE_ROOT}/libode.a)
elseif(ODE_ROOT)
  set(ODE_INC ${ODE_ROOT}/include)
  set(ODE_LIB ${ODE_ROOT}/lib/libode.a)
endif()
if(ODE_LIB)
  message(STATUS "ODE: ${ODE_LIB}")
  target_include_directories(eusviewode PRIVATE ${ODE_INC})
  target_link_libraries(eusviewode PRIVATE ${ODE_LIB})
  if(NOT APPLE)
    find_package(Threads)
    target_link_libraries(eusviewode PRIVATE Threads::Threads)
  endif()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ODE REQUIRED ode)
  message(STATUS "ODE (pkg-config): ${ODE_LINK_LIBRARIES} ${ODE_CFLAGS}")
  target_include_directories(eusviewode PRIVATE ${ODE_INCLUDE_DIRS})
  target_compile_options(eusviewode PRIVATE ${ODE_CFLAGS_OTHER})
  target_link_libraries(eusviewode PRIVATE ${ODE_LINK_LIBRARIES})
endif()
if(APPLE)
  set_target_properties(eusviewode PROPERTIES SUFFIX ".dylib")
endif()
